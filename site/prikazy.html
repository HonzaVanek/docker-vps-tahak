Příkazy na jedno místo

Když si potřebuješ rychle připomenout syntaxi bez dlouhého vysvětlování.

Linux

pwd
ls
ls -la

cd ..
cat soubor
nano soubor

Služby

systemctl status nginx

sudo systemctl reload nginx
sudo systemctl restart nginx

Síť

APT

sudo apt update
sudo apt install BALICEK

Docker

sudo docker ps
sudo docker ps -a
sudo docker image ls
sudo docker build -t app:v2 .
sudo docker stop app
sudo docker rm app
sudo docker logs app

Docker run

sudo docker run \
  --name app \
  --restart unless-stopped \
  -p 127.0.0.1:8080:80 \
  -d \
  app:v2

Nginx

sudo nano /etc/nginx/sites-available/app.example.cz
sudo ln -s /etc/nginx/sites-available/app.example.cz \
  /etc/nginx/sites-enabled/app.example.cz

sudo nginx -t
sudo systemctl reload nginx

Certbot

sudo certbot --nginx -d app.example.cz
systemctl status certbot.timer
sudo certbot renew --dry-run

Standardní deploy nové verze

cd ~/projekt
git pull

sudo docker build -t moje-app:vN .

sudo docker stop moje-app
sudo docker rm moje-app

sudo docker run \
  --name moje-app \
  --restart unless-stopped \
  -p 127.0.0.1:8080:80 \
  -d \
  moje-app:vN

sudo docker ps
